create extension if not exists pgcrypto;

-- RESET SEGURO DO PROJETO

drop trigger if exists on_auth_user_created_zarcovi on auth.users;
drop trigger if exists set_rpg_accounts_updated_at on public.rpg_accounts;
drop trigger if exists set_user_profiles_updated_at on public.user_profiles;

drop table if exists public.ryo_transactions cascade;
drop table if exists public.user_profiles cascade;
drop table if exists public.sync_logs cascade;
drop table if exists public.staff_members cascade;
drop table if exists public.rpg_accounts cascade;

drop function if exists public.handle_new_user() cascade;
drop function if exists public.set_updated_at() cascade;
drop function if exists public.is_staff_account(text) cascade;

-- FUNÇÕES

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

-- CONTAS DO RPG

create table public.rpg_accounts (
  id uuid primary key default gen_random_uuid(),
  account_code text unique not null,
  vila text,
  level bigint not null default 1,
  ryos bigint not null default 0,
  salary bigint not null default 0,
  cargos text,
  fire_will bigint not null default 0,
  stone_will bigint not null default 0,
  character_name text,
  treasure bigint not null default 0,
  ranking_position bigint,
  raw_data jsonb not null default '{}'::jsonb,
  source text not null default 'google_sheets',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint rpg_accounts_account_code_not_blank check (length(trim(account_code)) > 0)
);

create index rpg_accounts_account_code_idx on public.rpg_accounts (account_code);
create index rpg_accounts_vila_idx on public.rpg_accounts (vila);
create index rpg_accounts_level_idx on public.rpg_accounts (level desc);
create index rpg_accounts_ryos_idx on public.rpg_accounts (ryos desc);
create index rpg_accounts_ranking_position_idx on public.rpg_accounts (ranking_position);

create trigger set_rpg_accounts_updated_at
before update on public.rpg_accounts
for each row execute function public.set_updated_at();

-- PERFIS DE USUÁRIOS DO APP

create table public.user_profiles (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null unique references auth.users(id) on delete cascade,
  account_code text not null references public.rpg_accounts(account_code) on update cascade on delete restrict,
  display_name text,
  role text not null default 'user',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint user_profiles_role_check check (role in ('user', 'staff', 'admin', 'owner')),
  constraint user_profiles_account_code_not_blank check (length(trim(account_code)) > 0)
);

create index user_profiles_user_id_idx on public.user_profiles (user_id);
create index user_profiles_account_code_idx on public.user_profiles (account_code);
create index user_profiles_role_idx on public.user_profiles (role);

create trigger set_user_profiles_updated_at
before update on public.user_profiles
for each row execute function public.set_updated_at();

-- HISTÓRICO DE RYOS

create table public.ryo_transactions (
  id uuid primary key default gen_random_uuid(),
  account_code text not null references public.rpg_accounts(account_code) on update cascade on delete restrict,
  amount bigint not null,
  type text not null default 'adjustment',
  reason text,
  created_by uuid references auth.users(id) on delete set null,
  created_at timestamptz not null default now(),
  constraint ryo_transactions_type_check check (type in ('adjustment', 'reward', 'penalty', 'salary', 'sync', 'manual'))
);

create index ryo_transactions_account_code_idx on public.ryo_transactions (account_code);
create index ryo_transactions_created_at_idx on public.ryo_transactions (created_at desc);

-- LOGS DE SINCRONIZAÇÃO

create table public.sync_logs (
  id uuid primary key default gen_random_uuid(),
  status text not null,
  rows_imported bigint not null default 0,
  message text,
  created_at timestamptz not null default now(),
  constraint sync_logs_status_check check (status in ('success', 'error', 'warning'))
);

create index sync_logs_created_at_idx on public.sync_logs (created_at desc);

-- STAFF / ADMINS

create table public.staff_members (
  id uuid primary key default gen_random_uuid(),
  account_code text unique not null references public.rpg_accounts(account_code) on update cascade on delete cascade,
  role text not null default 'staff',
  created_at timestamptz not null default now(),
  constraint staff_members_role_check check (role in ('staff', 'admin', 'owner')),
  constraint staff_members_account_code_not_blank check (length(trim(account_code)) > 0)
);

create index staff_members_account_code_idx on public.staff_members (account_code);
create index staff_members_role_idx on public.staff_members (role);

create or replace function public.is_staff_account(check_account_code text)
returns boolean
language sql
stable
as $$
  select exists (
    select 1
    from public.staff_members
    where account_code = check_account_code
  );
$$;

-- RLS

alter table public.rpg_accounts enable row level security;
alter table public.user_profiles enable row level security;
alter table public.ryo_transactions enable row level security;
alter table public.sync_logs enable row level security;
alter table public.staff_members enable row level security;

-- POLÍTICAS
-- A API do Cloudflare usa SERVICE_ROLE_KEY para escrita administrativa.
-- Usuários logados só veem dados básicos necessários.

drop policy if exists "rpg_accounts_read_authenticated" on public.rpg_accounts;
create policy "rpg_accounts_read_authenticated"
on public.rpg_accounts
for select
to authenticated
using (true);

drop policy if exists "user_profiles_read_own" on public.user_profiles;
create policy "user_profiles_read_own"
on public.user_profiles
for select
to authenticated
using (user_id = auth.uid());

drop policy if exists "user_profiles_update_own_display" on public.user_profiles;
create policy "user_profiles_update_own_display"
on public.user_profiles
for update
to authenticated
using (user_id = auth.uid())
with check (user_id = auth.uid());

drop policy if exists "ryo_transactions_read_own" on public.ryo_transactions;
create policy "ryo_transactions_read_own"
on public.ryo_transactions
for select
to authenticated
using (
  account_code in (
    select account_code
    from public.user_profiles
    where user_id = auth.uid()
  )
);

drop policy if exists "staff_members_read_authenticated" on public.staff_members;
create policy "staff_members_read_authenticated"
on public.staff_members
for select
to authenticated
using (true);

drop policy if exists "sync_logs_no_direct_read" on public.sync_logs;
create policy "sync_logs_no_direct_read"
on public.sync_logs
for select
to authenticated
using (false);
